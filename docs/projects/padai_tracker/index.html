Survey Origins, Inferential Design & Analytical Scope

The Public Attitudes to Data and AI (PADAI) Tracker Survey, commissioned by the UK Department for Science, Innovation and Technology (DSIT) and the Responsible Technology Adoption (RTA) Unit and conducted by Savanta, monitors longitudinal shifts in how the UK public perceives data governance, institutional trustworthiness, and artificial intelligence across four waves (Wave 1 \(n=4,250\); Wave 2 \(n=4,320\); Wave 3 \(n=4,225\); Wave 4 \(n=4,947\)).

Rather than presenting static point estimates, this interactive dashboard surfaces statistical power, finite-sample volatility, and demographic interaction effects:

  1. Significance-Mapped Slopegraph: Evaluates structural shifts between Wave 1 (col a) and Wave 4 (col d). Line weight and stroke encoding are driven directly by two-tailed column proportion \(Z\)-tests (\(p < .05\)).
  2. Dynamic Uncertainty Ribbons: Computes finite-sample binomial standard errors (\(SE = \sqrt{p(1-p)/n}\)) natively in the browser, visibly expanding confidence intervals as you drill down from national aggregates (\(n \approx 4,947\)) to smaller regional or ethnic cohorts (\(n < 150\)).
  3. Longitudinal Cohort Transitions (Rosling Bubble Scatter): Animates joint trajectories of AI Awareness (Table 82) versus Institutional Trust & Data Utility (Table 1 / Table 22), weighted by subgroup sample size (\(n\)).
  4. Cross-Highlighted Distributional Ridge Plots: Compares full 4-point Likert trust distributions (Tables 3–12) across sectors and demographic strata with synchronized cross-panel hover highlighting.

slope_data = {
  const sub_rows = dataset.filter(
    d => d.Table_Name === selected_table_focus &&
         d.Demographic_Category === selected_demog_cat &&
         (d.Wave === "Wave 1" || d.Wave === "Wave 4") &&
         !["Total", "Unweighted Total", "Mean"].includes(d.Metric_Type)
  );

  // Group by Subgroup + Metric_Type to evaluate bidirectional significance (col 'a' vs col 'd')
  const grouped = d3.group(sub_rows, d => `${d.Subgroup}|||${d.Metric_Type}`);
  const pairs = [];

  for (const [key, rows] of grouped.entries()) {
    const w1 = rows.find(r => r.Wave === "Wave 1");
    const w4 = rows.find(r => r.Wave === "Wave 4");
    if (!w1 || !w4) continue;

    // p < .05 if Wave 1 beats Wave 4 ('d' in w1.beaten_letters) or Wave 4 beats Wave 1 ('a' in w4.beaten_letters)
    const is_sig = w1.beaten_letters.includes("d") || w4.beaten_letters.includes("a");
    const delta = w4.Value - w1.Value;
    const short_metric = w1.Metric_Type.replace(/\s*\(e\.g\..*?\)/g, "").trim();

    pairs.push(
      {
        Subgroup: w1.Subgroup,
        Metric_Type: short_metric,
        Wave: "Wave 1",
        wave_x: 1,
        Value: w1.Value,
        Unweighted_N: w1.Unweighted_N,
        Sig_Flag: w1.Significance_Flag,
        is_sig,
        delta,
        pair_id: key
      },
      {
        Subgroup: w4.Subgroup,
        Metric_Type: short_metric,
        Wave: "Wave 4",
        wave_x: 4,
        Value: w4.Value,
        Unweighted_N: w4.Unweighted_N,
        Sig_Flag: w4.Significance_Flag,
        is_sig,
        delta,
        pair_id: key
      }
    );
  }
  return pairs.filter(d => d.Subgroup === selected_subgroup);
}

Dynamic Uncertainty Ribbons: Sample Size & Standard Error Sensitivity 
ribbon_data = {
  const z_crit = z_lookup[ci_level];
  return dataset
    .filter(
      d => d.Table_Name === selected_table_focus &&
           d.Demographic_Category === selected_demog_cat &&
           d.Subgroup === selected_subgroup &&
           !["Total", "Unweighted Total", "Mean"].includes(d.Metric_Type)
    )
    .map(d => {
      const p = d.Value;
      const n = d.Unweighted_N;
      // Native Binomial Standard Error: SE = sqrt(p * (1 - p) / n)
      const se = n > 0 ? Math.sqrt((p * (1 - p)) / n) : 0;
      const ci_low = Math.max(0, p - z_crit * se);
      const ci_high = Math.min(1, p + z_crit * se);
      return {
        ...d,
        short_metric: d.Metric_Type.replace(/\s*\(e\.g\..*?\)/g, "").trim(),
        se,
        ci_low,
        ci_high,
        half_width_pp: z_crit * se * 100
      };
    });
}

mean_sample_n = d3.mean(ribbon_data, d => d.Unweighted_N) || 0
mean_moe_pp = d3.mean(ribbon_data, d => d.half_width_pp) || 0

rosling_trajectories = {
  // X-axis: Table 82 ' NET: Could explain at least partially ' (or 'NET: Heard of ')
  const x_rows = dataset.filter(
    d => d.Table_Name === "Table 82" &&
         d.Metric_Type.trim() === "NET: Could explain at least partially" &&
         d.Demographic_Category !== "Digital engagement"
  );
  // Y-axis: Table 1 selected organisation Net Trust
  const y_rows = dataset.filter(
    d => d.Table_Name === "Table 1" &&
         d.Metric_Type.trim() === y_trust_target &&
         d.Demographic_Category !== "Digital engagement"
  );

  const y_map = new Map(
    y_rows.map(d => [`${d.Demographic_Category}|${d.Subgroup}|${d.wave_num}`, d])
  );

  const joined = [];
  for (const x of x_rows) {
    const key = `${x.Demographic_Category}|${x.Subgroup}|${x.wave_num}`;
    const y = y_map.get(key);
    if (y) {
      joined.push({
        Demographic_Category: x.Demographic_Category,
        Subgroup: x.Subgroup,
        wave_num: x.wave_num,
        ai_awareness: x.Value,
        net_trust: y.Value,
        Unweighted_N: x.Unweighted_N
      });
    }
  }

  // Interpolate smoothly at current wave_scrubber position
  const by_cohort = d3.group(joined, d => `${d.Demographic_Category}|${d.Subgroup}`);
  const current_points = [];
  const trail_segments = [];

  const w_floor = Math.floor(wave_scrubber);
  const w_ceil = Math.min(4, Math.ceil(wave_scrubber));
  const alpha = wave_scrubber - w_floor;

  for (const [cohort_key, pts] of by_cohort.entries()) {
    pts.sort((a, b) => a.wave_num - b.wave_num);
    for (const pt of pts) {
      if (pt.wave_num <= wave_scrubber) {
        trail_segments.push({ ...pt, cohort_key });
      }
    }
    const p0 = pts.find(d => d.wave_num === w_floor);
    const p1 = pts.find(d => d.wave_num === w_ceil) || p0;
    if (p0 && p1) {
      const interp_x = p0.ai_awareness + alpha * (p1.ai_awareness - p0.ai_awareness);
      const interp_y = p0.net_trust + alpha * (p1.net_trust - p0.net_trust);
      const interp_n = Math.round(p0.Unweighted_N + alpha * (p1.Unweighted_N - p0.Unweighted_N));
      current_points.push({
        cohort_key,
        Demographic_Category: p0.Demographic_Category,
        Subgroup: p0.Subgroup,
        ai_awareness: interp_x,
        net_trust: interp_y,
        Unweighted_N: interp_n
      });
      trail_segments.push({
        cohort_key,
        Demographic_Category: p0.Demographic_Category,
        Subgroup: p0.Subgroup,
        wave_num: wave_scrubber,
        ai_awareness: interp_x,
        net_trust: interp_y,
        Unweighted_N: interp_n
      });
    }
  }
  return { current_points, trail_segments };
}

sector_table_map = new Map([
  ["Table 3", "The NHS"],
  ["Table 5", "Academic Researchers"],
  ["Table 12", "Banks & Finance"],
  ["Table 7", "Big Tech Companies"],
  ["Table 4", "The Government"],

likert_points = new Map([
  ["Do not trust at all (1.0)", 1.0],
  ["Do not trust much (2.0)", 2.0],
  ["Somewhat trust (3.0)", 3.0],
  ["Trust a lot (4.0)", 4.0]

// Smooth continuous density interpolation across the 4 Likert anchor points for ridgelines
ridge_density_data = {
  const raw_likert = dataset.filter(
    d => sector_table_map.has(d.Table_Name) &&
         d.Wave === "Wave 4" &&
         d.Demographic_Category === selected_demog_cat &&
         likert_points.has(d.Metric_Type)
  );

  const grouped = d3.group(raw_likert, d => `${d.Table_Name}|${d.Subgroup}`);
  const grid_x = d3.range(1.0, 4.05, 0.1);
  const curves = [];

  for (const [key, rows] of grouped.entries()) {
    const [t_name, subgroup] = key.split("|");
    const sector = sector_table_map.get(t_name);
    const weights = rows.map(r => ({
      score: likert_points.get(r.Metric_Type),
      prob: r.Value,
      n: r.Unweighted_N
    }));

    // Gaussian kernel smoothing (bandwidth = 0.42) over discrete Likert mass
    const bw = 0.42;
    for (const x of grid_x) {
      let density = 0;
      for (const w of weights) {
        const u = (x - w.score) / bw;
        density += w.prob * Math.exp(-0.5 * u * u);
      }
      curves.push({
        Sector: sector,
        Subgroup: subgroup,
        likert_x: Number(x.toFixed(2)),
        density,
        Unweighted_N: weights[0]?.n || 0
      });
    }
  }
  return curves;
}
